list(SORT NOLF_SDK_ABI_INPUTS)
set(NOLF_SDK_ABI_CONTENT "")
foreach(NOLF_SDK_INPUT IN LISTS NOLF_SDK_ABI_INPUTS)
  if(NOT IS_ABSOLUTE "${NOLF_SDK_INPUT}")
    set(NOLF_SDK_INPUT "${CMAKE_CURRENT_SOURCE_DIR}/${NOLF_SDK_INPUT}")
  endif()
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS "${NOLF_SDK_INPUT}")
  file(SHA256 "${NOLF_SDK_INPUT}" NOLF_SDK_FILE_HASH)
  file(RELATIVE_PATH NOLF_SDK_RELATIVE "${CMAKE_CURRENT_SOURCE_DIR}" "${NOLF_SDK_INPUT}")
  string(APPEND NOLF_SDK_ABI_CONTENT "${NOLF_SDK_RELATIVE}:${NOLF_SDK_FILE_HASH};")
endforeach()
string(SHA256 NOLF_SDK_ABI_HASH "${NOLF_SDK_ABI_CONTENT}")
string(SUBSTRING "${NOLF_SDK_ABI_HASH}" 0 16 NOLF_SDK_ABI_HASH)

if(MSVC)
  # The legacy SDK still accepts string literals through char* entry points.
  target_compile_options(nolf_options INTERFACE /W4 /permissive- /Zc:strictStrings- /utf-8)
else()
  # The engine and game sources were written for MSVC, which never optimises
  # on type-based aliasing, and still read objects through layout views.
  target_compile_options(nolf_options INTERFACE -Wall -Wextra -Wpedantic -fno-strict-aliasing)
  if(NOLF_SANITIZERS)
    target_compile_options(nolf_options INTERFACE -fsanitize=address,undefined -fno-sanitize-recover=undefined -fno-omit-frame-pointer)
    target_link_options(nolf_options INTERFACE -fsanitize=address,undefined)
    if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
      # Clang otherwise leaves sanitizer symbols for the executable to
      # supply, which cannot satisfy the strict shared-library link gate.
      target_link_options(nolf_options INTERFACE -shared-libsan)
      if(UNIX)
        execute_process(COMMAND "${CMAKE_CXX_COMPILER}" -print-runtime-dir
          OUTPUT_VARIABLE NOLF_SANITIZER_RUNTIME_DIR OUTPUT_STRIP_TRAILING_WHITESPACE COMMAND_ERROR_IS_FATAL ANY)
        target_link_options(nolf_options INTERFACE "LINKER:-rpath,${NOLF_SANITIZER_RUNTIME_DIR}")
      endif()
    endif()
  endif()
endif()
